// supabase/functions/generate-questions/index.ts // // Deploy with: // supabase functions deploy generate-questions // // Expects the frontend to POST: // { // source: string, // notes or a topic, e.g. "the French Revolution" (max 4000 chars, longer is truncated) // count: number, // how many questions to generate (1–10, capped server-side to limit credit use) // images: string[] // optional array of data URLs (max 2 kept, extras are dropped) // } // // Requires a logged-in Supabase user — the request must include // "Authorization: Bearer " (the frontend already does this). // // The OpenRouter API key is NOT sent by the client. It's read from a // server-side secret, set once with: // supabase secrets set OPENROUTER_API_KEY=sk-or-v1-... // This means visitors never see the key, but all usage bills to your account. // // Optionally set OPENROUTER_MODEL to pick a different model. Defaults to // Claude Haiku via OpenRouter, which is much cheaper per token than Sonnet/Opus // while still handling this task well. Vision-capable models are required if // you want image attachments to work. // // Returns: // { questions: [ { question: string, options: string[], correctIndex: number }, ... ] } import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const OPENROUTER_MODEL = Deno.env.get("OPENROUTER_MODEL") || "anthropic/claude-3.5-haiku"; const OPENROUTER_API_URL = "https://openrouter.ai/api/v1/chat/completions"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; function jsonResponse(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); } Deno.serve(async (req: Request) => { // Handle CORS preflight if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return jsonResponse({ error: "Method not allowed" }, 405); } try { // --- 1. Verify the caller is a logged-in Supabase user --- const authHeader = req.headers.get("Authorization") ?? ""; const jwt = authHeader.replace(/^Bearer\s+/i, ""); if (!jwt) { return jsonResponse({ error: "You need to be logged in to generate questions." }, 401); } // SUPABASE_URL and SUPABASE_ANON_KEY are auto-provided by the Supabase // Edge Runtime for every function — no need to set them as secrets yourself. const supabaseUrl = Deno.env.get("SUPABASE_URL")!; const supabaseAnonKey = Deno.env.get("SUPABASE_ANON_KEY")!; // Validate the JWT belongs to a real user const supabase = createClient(supabaseUrl, supabaseAnonKey, { global: { headers: { Authorization: `Bearer ${jwt}` } }, }); const { data: userData, error: userError } = await supabase.auth.getUser(); if (userError || !userData?.user) { return jsonResponse({ error: "You need to be logged in to generate questions." }, 401); } // --- 2. Parse and validate the request body --- const body = await req.json().catch(() => null); if (!body) { return jsonResponse({ error: "Invalid JSON body." }, 400); } const source: string = typeof body.source === "string" ? body.source.slice(0, 4000) : ""; const count: number = Number.isFinite(body.count) ? Math.min(Math.max(body.count, 1), 10) : 6; const images: string[] = Array.isArray(body.images) ? body.images.slice(0, 2) : []; if (!source && images.length === 0) { return jsonResponse({ error: "Please provide some notes, a topic, or an image." }, 400); } // API key lives server-side only — set via `supabase secrets set OPENROUTER_API_KEY=...` const apiKey = Deno.env.get("OPENROUTER_API_KEY"); if (!apiKey) { console.error("OPENROUTER_API_KEY secret is not set."); return jsonResponse({ error: "Server is not configured. Please contact the site owner." }, 500); } // --- 3. Build the OpenRouter chat completion request --- const instructions = `Generate exactly ${count} multiple-choice questions from the study material below (notes/topic text and/or attached images). Rules: - Exactly 4 options per question, exactly 1 correct. - Vary the position of the correct answer. - Clear, unambiguous, answerable from the material. No trick questions. Respond with ONLY valid JSON, no markdown fences, no commentary: {"questions":[{"question":"string","options":["string","string","string","string"],"correctIndex":0}]}`; const contentBlocks: Record[] = []; if (source) { contentBlocks.push({ type: "text", text: `Study material:\n${source}` }); } for (const dataUrl of images) { const match = /^data:(image\/[a-zA-Z0-9.+-]+);base64,(.+)$/.exec(dataUrl); if (!match) continue; // skip anything that isn't a valid data URL contentBlocks.push({ type: "image_url", image_url: { url: dataUrl }, }); } if (contentBlocks.length === 0) { return jsonResponse({ error: "No usable study material was provided." }, 400); } const openRouterResponse = await fetch(OPENROUTER_API_URL, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${apiKey}`, // Optional but recommended by OpenRouter for attribution/rankings "HTTP-Referer": supabaseUrl, "X-Title": "Confidence AI", }, body: JSON.stringify({ model: OPENROUTER_MODEL, max_tokens: Math.min(1000, 120 + count * 70), messages: [ { role: "system", content: instructions }, { role: "user", content: contentBlocks }, ], }), }); if (!openRouterResponse.ok) { const errText = await openRouterResponse.text().catch(() => ""); console.error("OpenRouter API error:", openRouterResponse.status, errText); if (openRouterResponse.status === 401) { return jsonResponse({ error: "Server's API key was rejected. Contact the site owner." }, 401); } if (openRouterResponse.status === 402) { return jsonResponse( { error: "The site has run out of AI credits. Please contact the site owner to top up." }, 402, ); } if (openRouterResponse.status === 429) { return jsonResponse({ error: "Too many requests right now. Please try again in a moment." }, 429); } return jsonResponse({ error: "Failed to generate questions. Please try again." }, 502); } const openRouterData = await openRouterResponse.json(); const rawText: string = openRouterData?.choices?.[0]?.message?.content ?? ""; // Strip accidental markdown fences just in case const cleaned = rawText.replace(/^```json\s*|^```\s*|```\s*$/gm, "").trim(); let parsed: { questions?: unknown }; try { parsed = JSON.parse(cleaned); } catch (_e) { console.error("Failed to parse OpenRouter response:", rawText); return jsonResponse({ error: "Could not parse generated questions. Please try again." }, 502); } const questions = Array.isArray(parsed.questions) ? parsed.questions : []; // --- 4. Validate the shape before returning --- const validQuestions = questions.filter((q: unknown) => { const question = q as { question?: unknown; options?: unknown; correctIndex?: unknown }; return ( typeof question.question === "string" && Array.isArray(question.options) && question.options.length === 4 && question.options.every((o: unknown) => typeof o === "string") && typeof question.correctIndex === "number" && question.correctIndex >= 0 && question.correctIndex <= 3 ); }); if (validQuestions.length === 0) { return jsonResponse({ error: "No valid questions were generated. Please try again." }, 502); } return jsonResponse({ questions: validQuestions }); } catch (err) { console.error("Unexpected error:", err); return jsonResponse({ error: "Something went wrong generating questions." }, 500); } });
